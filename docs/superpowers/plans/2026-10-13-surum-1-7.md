```swift
import Foundation

/// Hedef donemi. Hesap her zaman aylik yapilir (sabit ucret ve aylik kademeler ay bazinda); gun/hafta aya cevrilir.
enum GoalPeriod: String, Codable, CaseIterable, Identifiable {
    case day, week, month
    var id: String { rawValue }
    var displayName: String {
        switch self {
        case .day: "Günlük"
        case .week: "Haftalık"
        case .month: "Aylık"
        }
    }
}

/// Kullanicinin net (yakit dusulmus) kazanc hedefi.
struct GoalSpec: Codable, Equatable {
    var amount: Money
    var period: GoalPeriod
    var daysPerWeek: Int = 6
    var maxHoursPerDay: Decimal = 10

    init(amount: Money, period: GoalPeriod, daysPerWeek: Int = 6, maxHoursPerDay: Decimal = 10) {
        self.amount = amount
        self.period = period
        self.daysPerWeek = min(7, max(1, daysPerWeek))
        self.maxHoursPerDay = maxHoursPerDay
    }

    /// Ayda calisilan gun: haftalik gun × 52 / 12, en yakin tam sayi (6 -> 26).
    var monthWorkDays: Int { GoalSpec.monthWorkDays(daysPerWeek: daysPerWeek) }

    static func monthWorkDays(daysPerWeek: Int) -> Int {
        let value: Double = Double(daysPerWeek) * 52.0 / 12.0
        return Int(value.rounded())
    }

    /// Hedefin aylik karsiligi.
    func monthlyTarget() -> Money {
        switch period {
        case .month: return amount
        case .week: return Money(Money.round(amount.value * Decimal(monthWorkDays) / Decimal(daysPerWeek)))
        case .day: return amount * monthWorkDays
        }
    }

    /// Hedef donemindeki is gunu.
    func periodWorkDays() -> Int {
        switch period {
        case .day: 1
        case .week: daysPerWeek
        case .month: monthWorkDays
        }
    }
}

/// Secilen firmanin tarifesinin ihtiyac duydugu kisisel bilgiler.
struct CourierTariffProfile: Equatable {
    var platformCode: String
    var regionTier: String?
    var seniorityYears: Int?
    var vehicleType: String?
    var flags: [String]
}

enum FuelType: String, Codable, CaseIterable, Identifiable {
    case gasoline, diesel, electric
    var id: String { rawValue }
    var displayName: String {
        switch self {
        case .gasoline: "Benzin"
        case .diesel: "Motorin"
        case .electric: "Elektrik"
        }
    }
    var unitName: String { self == .electric ? "kWh" : "L" }
}

enum FuelPriceSource: String, Codable {
    case manual, server
}

/// Tarihli yakit fiyati. Simdilik elle girilir; `server` ileride sunucudan gelecek fiyatlar icin.
struct FuelPriceEntry: Codable, Equatable, Identifiable {
    var id: UUID
    var day: String
    var pricePerUnit: Money
    var type: FuelType
    var source: FuelPriceSource

    init(id: UUID = UUID(), day: String, pricePerUnit: Money, type: FuelType, source: FuelPriceSource = .manual) {
        self.id = id
        self.day = day
        self.pricePerUnit = pricePerUnit
        self.type = type
        self.source = source
    }
}

struct FuelSettings: Codable, Equatable {
    var entries: [FuelPriceEntry] = []
    /// L (ya da kWh) / 100 km. nil: arac turunun varsayilani.
    var consumptionPer100Km: Decimal? = nil
}
```

```swift
import Foundation

/// Yakit gideri: km × tuketim (L/100 km) × o gunun fiyati. Fiyat hic girilmemisse eski km basi maliyet (yedek).
struct FuelCostModel: Equatable {
    var settings: FuelSettings
    var vehicleType: String?
    var fallbackCostPerKm: Money

    init(settings: FuelSettings, vehicleType: String?, fallbackCostPerKm: Money) {
        self.settings = settings
        self.vehicleType = vehicleType
        self.fallbackCostPerKm = fallbackCostPerKm
    }

    static func defaultConsumption(vehicleType: String?) -> Decimal {
        vehicleType == "car" ? Decimal(string: "6.5")! : 3
    }

    var consumption: Decimal { settings.consumptionPer100Km ?? FuelCostModel.defaultConsumption(vehicleType: vehicleType) }

    var hasPrice: Bool { !settings.entries.isEmpty }

    private var sorted: [FuelPriceEntry] { settings.entries.sorted { $0.day < $1.day } }

    /// Gunden once ya da o gun girilmis en yeni fiyat; o gunden once hic yoksa en eski fiyat.
    func price(on day: String) -> FuelPriceEntry? {
        let entries: [FuelPriceEntry] = sorted
        return entries.last { $0.day <= day } ?? entries.first
    }

    func costPerKm(on day: String) -> Money {
        guard let entry = price(on: day) else { return fallbackCostPerKm }
        return Money(entry.pricePerUnit.value * consumption / 100)
    }

    func cost(km: Decimal, on day: String) -> Money {
        Money(Money.round(costPerKm(on: day).value * km))
    }

    /// Son fiyat `maxAgeDays` gunden eskiyse (ya da hic yoksa) guncellenmeli.
    func isStale(today: String, maxAgeDays: Int = 7) -> Bool {
        guard let last = sorted.last else { return true }
        guard let age = RuleDay.days(from: last.day, to: today) else { return true }
        return age > maxAgeDays
    }

    /// Iki gun arasindaki fiyat degisimi (yuzde, 2 hane). Fiyat yoksa nil.
    func change(from: String, to: String) -> Decimal? {
        guard let old = price(on: from), let new = price(on: to), old.pricePerUnit.value > 0 else { return nil }
        return Money.round((new.pricePerUnit.value - old.pricePerUnit.value) / old.pricePerUnit.value * 100)
    }
}
```

```swift
struct TariffQuestions: Equatable {
    var regionTiers: [String] = []
    var asksSeniority: Bool = false
    var flags: [String] = []
    var vehicleTypes: [String] = []

    var isEmpty: Bool { regionTiers.isEmpty && !asksSeniority && flags.isEmpty && vehicleTypes.isEmpty }

        let versions: [TariffVersion] = catalog.usable(platform: platform, day: day)
        var tiers: Set<String> = []
        var flags: Set<String> = []
        var seniority = false
        var vehicles: Set<String> = []
        for version in versions {
            if let vehicle = version.vehicleType { vehicles.insert(vehicle) }
            for item in version.activeItems(on: day) {
                let c = item.conditions
                if let tier = c.regionTier { tiers.insert(tier) }
                if let flag = c.requiresFlag { flags.insert(flag) }
                if c.minSeniorityYears != nil || c.maxSeniorityYears != nil { seniority = true }
            }
        }
        return TariffQuestions(regionTiers: tiers.sorted(), asksSeniority: seniority, flags: flags.sorted(),
                               vehicleTypes: vehicles.sorted())
    }

    static func regionName(_ tier: String) -> String {
        switch tier {

        }

        switch flag {
        case "woman_employment": "Kadın istihdam desteğinden yararlanıyorum"

        }
    }

    static func vehicleName(_ type: String) -> String {
```

```swift
import Foundation

/// Kuryenin secili firmadaki hizi: son 8 haftanin kayitlarindan paket/saat ve km/paket.
/// En az 5 paketli kayit ve 10 saat yoksa varsayilan kullanilir ve `isAssumed` isaretlenir.
struct PaceEstimate: Equatable {
    var packagesPerHour: Decimal
    var kmPerPackage: Decimal
    var isAssumed: Bool
    var sampleCount: Int

    static let defaultPackagesPerHour: Decimal = Decimal(string: "2.5")!
    static let defaultKmPerPackage: Decimal = Decimal(string: "3.5")!
    static let windowDays: Double = 56
    static let minimumSamples = 5
    static let minimumHours: Decimal = 10

    static func estimate(entries: [EarningEntry], platform: String, now: Date,
                         defaults: (packagesPerHour: Decimal, kmPerPackage: Decimal)? = nil) -> PaceEstimate {
        let fallbackRate: Decimal = defaults?.packagesPerHour ?? defaultPackagesPerHour
        let fallbackKm: Decimal = defaults?.kmPerPackage ?? defaultKmPerPackage
        let cutoff: Date = now.addingTimeInterval(-windowDays * 86_400)
        let usable: [EarningEntry] = entries.filter { entry in
            guard entry.platformCode == platform, entry.start >= cutoff, entry.end > entry.start else { return false }
            guard let packages = entry.packageCount else { return false }
            return packages > 0
        }
        var packages: Decimal = 0
        var hours: Decimal = 0
        var km: Decimal = 0
        for entry in usable {
            packages += Decimal(entry.packageCount ?? 0)
            hours += Decimal(entry.end.timeIntervalSince(entry.start) / 3600)
            km += Decimal(entry.distanceKm)
        }
        guard usable.count >= minimumSamples, hours >= minimumHours, packages > 0 else {
            return PaceEstimate(packagesPerHour: fallbackRate, kmPerPackage: fallbackKm, isAssumed: true, sampleCount: usable.count)
        }
        let rate: Decimal = Money.round(packages / hours, scale: 1)
        let kmPer: Decimal = km > 0 ? Money.round(km / packages, scale: 1) : fallbackKm
        return PaceEstimate(packagesPerHour: rate, kmPerPackage: kmPer, isAssumed: false, sampleCount: usable.count)
    }
}
```

Calculation notes for the implementer:
- Always a **month** `PayInput` (`span: .month`, `startDay` = first day of `day`'s month, `endDay` = last day), `dayPackages = N`, `periodPackages = N`, `totalKm = N × kmPerPackage`, `hours = N / packagesPerHour` (2 decimals), `regionTier/seniorityYears/vehicleType/flags` from profile; Getir without vehicle → `vehicleType = "motorcycle"`.
- `sideIncome` = non-cash side benefits whose item has `requiresFlag` contained in `profile.flags` and `unit == .month` (user decision: included in goal).
- `net = gross + sideIncome − fuel`, `fuel = fuelModel.cost(km: totalKm, on: day)`.
- Search smallest `N ∈ [0, 20000]` with `net(N) ≥ monthlyTarget` (binary search; `net` is non-decreasing in N for all four calculators).
- `alreadyMet` when `net(0) ≥ target`.
- Capacity: `maxPerDay = floor(packagesPerHour × maxHoursPerDay)`; if `ceil(N / workDays) > maxPerDay` → try `daysPerWeek+1 … 7` (month work days via `GoalSpec.monthWorkDays`); first one where `net(maxPerDay × days') ≥ target` → `needsMoreDays(d)`; else `unreachable(maxNet: net(maxPerDay × monthWorkDays))`. For `.day` period the extra-days search is skipped (always `unreachable`).
- `hoursPerDay` = `packagesPerDay / packagesPerHour` rounded **up** to 0.5.
- `otherTierDelta`: if `profile.regionTier != nil` and the breakdown has an included `.fixed` line whose item has `regionTier`, compute same N with the other tier(s); delta = this gross − other gross (max abs among others).
- `nextTier`: from `breakdown.progress` the first entry with `unit == .month` whose `target > N` and `target − N ≤ ceil(N × 0.2)`; extraNet = `net(target) − net(N)`; newRate = progress reward.
- `guaranteed` = included cash `.fixed` + `.hourly` lines + `sideIncome` lines (isSideBenefit = true).
- `packageIncome = gross − Σ(fixed + hourly lines)`.

```swift
import Foundation

struct GoalLine: Equatable, Identifiable {
    var id: String
    var label: String
    var amount: Money
    var isSideBenefit: Bool
    var confidence: ConfidenceStatus
}

enum GoalReachability: Equatable {
    /// Paketsiz gelir (sabit ucret + yan haklar) hedefi zaten karsiliyor.
    case alreadyMet
    case reachable
    /// Haftada bu kadar gun calisilirsa ulasilir.
    case needsMoreDays(Int)
    /// Saat ve gun sinirinda ulasilabilecek en yuksek aylik net.
    case unreachable(maxNet: Money)
}

struct NextTierHint: Equatable {
    var threshold: Int
    var extraPackages: Int
    var extraNet: Money
    var newRate: Money
}

/// Hedef plani: secili firmanin tarifesi tersine calistirilarak bulunan en az paket / saat.
struct GoalPlan: Equatable {
    var platformCode: String
    var goal: GoalSpec
    var monthlyPackages: Int
    var packagesPerDay: Int
    var hoursPerDay: Decimal
    var workDays: Int
    var guaranteed: [GoalLine]
    var guaranteedTotal: Money
    var packageIncome: Money
    var gross: Money
    var sideIncome: Money
    var fuel: Money
    var net: Money
    var regionTier: String?
    var otherTierDelta: Money?
    var nextTier: NextTierHint?
    var pace: PaceEstimate
    var reachability: GoalReachability
    var usesFuelFallback: Bool
    var breakdown: PayBreakdown

    /// Hedef donemindeki paket (haftalik/gunluk hedefte aylik plandan orantili, yukari yuvarlanir).
    func periodPackages() -> Int {
        guard goal.period != .month, workDays > 0 else { return monthlyPackages }
        let share: Double = Double(monthlyPackages) * Double(goal.periodWorkDays()) / Double(workDays)
        return Int(share.rounded(.up))
    }

    /// Hedef donemindeki net (hedef tutariyla ayni olcek).
    func periodNet() -> Money {
        guard goal.period != .month, workDays > 0 else { return net }
        let target: Money = goal.monthlyTarget()
        guard target.value > 0 else { return net }
        return Money(Money.round(goal.amount.value * net.value / target.value))
    }
}

struct GoalPlanner {
    var engine: PayEngine
    var fuel: FuelCostModel
    static let maxPackages = 20_000

    struct Evaluation: Equatable {
        var breakdown: PayBreakdown
        var gross: Money
        var sideLines: [PayLine]
        var sideIncome: Money
        var fuel: Money
        var net: Money
        var hours: Decimal
    }

    init(engine: PayEngine, fuel: FuelCostModel) {
        self.engine = engine
        self.fuel = fuel
    }

    /// Kullanicinin isaretini tasidigi, aylik, nakit olmayan yan haklar (or. kadin istihdam destegi). Kullanici karariyla
    /// hedefe dahil edilir; dogrulanmamissa sayfa altindaki notta listelenir.
    static func flaggedSideIncome(_ breakdown: PayBreakdown, version: TariffVersion?, flags: [String]) -> [PayLine] {
        guard let version else { return [] }
        return breakdown.sideBenefits.filter { line in
            guard let itemID = line.itemID, let item = version.items.first(where: { $0.id == itemID }) else { return false }
            guard let flag = item.conditions.requiresFlag, flags.contains(flag) else { return false }
            return item.unit == .month && line.amount.value > 0
        }
    }

    static func monthBounds(of day: String) -> (start: String, end: String) {
        let parts: [Int] = day.split(separator: "-").compactMap { Int($0) }
        guard parts.count == 3 else { return (day, day) }
        let year = parts[0], month = parts[1]
        let lengths = [31, (year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)) ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
        let last: Int = lengths[max(0, min(11, month - 1))]
        return (String(format: "%04d-%02d-01", year, month), String(format: "%04d-%02d-%02d", year, month, last))
    }

    func input(packages: Int, profile: CourierTariffProfile, pace: PaceEstimate, day: String) -> PayInput {
        let bounds = GoalPlanner.monthBounds(of: day)
        var input = PayInput(platformCode: profile.platformCode, span: .month, startDay: bounds.start, endDay: bounds.end)
        input.dayPackages = packages
        input.periodPackages = packages
        input.totalKm = Decimal(packages) * pace.kmPerPackage
        input.hours = pace.packagesPerHour > 0 ? Money.round(Decimal(packages) / pace.packagesPerHour) : 0
        input.regionTier = profile.regionTier
        input.seniorityYears = profile.seniorityYears
        input.vehicleType = profile.vehicleType ?? (profile.platformCode == "getir" ? "motorcycle" : nil)
        input.flags = profile.flags
        return input
    }

    func evaluate(packages: Int, profile: CourierTariffProfile, pace: PaceEstimate, day: String) -> Result<Evaluation, PayError> {
        let payInput: PayInput = input(packages: packages, profile: profile, pace: pace, day: day)
        let breakdown: PayBreakdown
        switch engine.calculate(payInput) {
        case let .success(value): breakdown = value
        case let .failure(error): return .failure(error)
        }
        let version: TariffVersion? = engine.catalog.versions.first { $0.id == breakdown.tariffVersionID }
        let side: [PayLine] = GoalPlanner.flaggedSideIncome(breakdown, version: version, flags: profile.flags)
        let sideIncome: Money = side.map(\.amount).sum()
        let fuelCost: Money = fuel.cost(km: payInput.totalKm ?? 0, on: day)
        let net: Money = breakdown.gross + sideIncome - fuelCost
        return .success(Evaluation(breakdown: breakdown, gross: breakdown.gross, sideLines: side, sideIncome: sideIncome,
                                   fuel: fuelCost, net: net, hours: payInput.hours))
    }

    private func net(_ packages: Int, _ profile: CourierTariffProfile, _ pace: PaceEstimate, _ day: String) -> Money? {
        if case let .success(e) = evaluate(packages: packages, profile: profile, pace: pace, day: day) { return e.net }
        return nil
    }

    /// net(N) >= target olan en kucuk N (yoksa nil).
    private func smallestPackages(target: Money, upTo limit: Int, _ profile: CourierTariffProfile, _ pace: PaceEstimate,
                                  _ day: String) -> Int? {
        guard let top = net(limit, profile, pace, day), !(top < target) else { return nil }
        var low = 0, high = limit
        while low < high {
            let mid: Int = (low + high) / 2
            if let value = net(mid, profile, pace, day), !(value < target) { high = mid } else { low = mid + 1 }
        }
        return low
    }

    func plan(goal: GoalSpec, profile: CourierTariffProfile, pace: PaceEstimate, day: String) -> Result<GoalPlan, PayError> {
        let target: Money = goal.monthlyTarget()
        let workDays: Int = goal.monthWorkDays
        guard case let .success(zero) = evaluate(packages: 0, profile: profile, pace: pace, day: day) else {
            if case let .failure(error) = evaluate(packages: 0, profile: profile, pace: pace, day: day) { return .failure(error) }
            return .failure(.unsupportedPlatform(profile.platformCode))
        }
        let maxPerDay: Int = NSDecimalNumber(decimal: pace.packagesPerHour * goal.maxHoursPerDay).intValue
        let capacity: Int = maxPerDay * workDays

        var packages: Int
        var reachability: GoalReachability
        if !(zero.net < target) {
            packages = 0
            reachability = .alreadyMet
        } else if let needed = smallestPackages(target: target, upTo: GoalPlanner.maxPackages, profile, pace, day),
                  needed <= capacity {
            packages = needed
            reachability = .reachable
        } else {
            packages = capacity
            reachability = .unreachable(maxNet: net(capacity, profile, pace, day) ?? zero.net)
            if goal.period != .day, goal.daysPerWeek < 7 {
                for days in (goal.daysPerWeek + 1)...7 {
                    let more: Int = maxPerDay * GoalSpec.monthWorkDays(daysPerWeek: days)
                    if let value = net(more, profile, pace, day), !(value < target) {
                        reachability = .needsMoreDays(days)
                        break
                    }
                }
            }
        }

        guard case let .success(result) = evaluate(packages: packages, profile: profile, pace: pace, day: day) else {
            return .failure(.unsupportedPlatform(profile.platformCode))
        }
        let perDay: Int = workDays > 0 ? Int((Double(packages) / Double(workDays)).rounded(.up)) : packages
        let rawHours: Decimal = pace.packagesPerHour > 0 ? Decimal(perDay) / pace.packagesPerHour : 0
        let hoursPerDay: Decimal = GoalPlanner.roundUpToHalf(rawHours)

        let base: [PayLine] = result.breakdown.lines.filter { $0.included && $0.isCash && ($0.type == .fixed || $0.type == .hourly) }
        var guaranteed: [GoalLine] = base.map {
            GoalLine(id: $0.id, label: $0.label, amount: $0.amount, isSideBenefit: false, confidence: $0.confidence)
        }
        guaranteed += result.sideLines.map {
            GoalLine(id: $0.id, label: $0.label, amount: $0.amount, isSideBenefit: true, confidence: $0.confidence)
        }
        let baseTotal: Money = base.map(\.amount).sum()

        return .success(GoalPlan(
            platformCode: profile.platformCode, goal: goal, monthlyPackages: packages, packagesPerDay: perDay,
            hoursPerDay: hoursPerDay, workDays: workDays, guaranteed: guaranteed,
            guaranteedTotal: baseTotal + result.sideIncome, packageIncome: result.gross - baseTotal, gross: result.gross,
            sideIncome: result.sideIncome, fuel: result.fuel, net: result.net, regionTier: profile.regionTier,
            otherTierDelta: otherTierDelta(packages: packages, profile: profile, pace: pace, day: day, gross: result.gross,
                                           breakdown: result.breakdown),
            nextTier: nextTier(packages: packages, result: result, profile: profile, pace: pace, day: day),
            pace: pace, reachability: reachability, usesFuelFallback: !fuel.hasPrice, breakdown: result.breakdown
        ))
    }

    static func roundUpToHalf(_ value: Decimal) -> Decimal {
        let doubled: Double = NSDecimalNumber(decimal: value * 2).doubleValue
        return Decimal(doubled.rounded(.up)) / 2
    }

    private func otherTierDelta(packages: Int, profile: CourierTariffProfile, pace: PaceEstimate, day: String, gross: Money,
                                breakdown: PayBreakdown) -> Money? {
        guard let tier = profile.regionTier,
              let version = engine.catalog.versions.first(where: { $0.id == breakdown.tariffVersionID }) else { return nil }
        let tiers: Set<String> = Set(version.items.compactMap { $0.type == .fixed ? $0.conditions.regionTier : nil })
        guard tiers.contains(tier), tiers.count > 1 else { return nil }
        var best: Money? = nil
        for other in tiers.sorted() where other != tier {
            var alt = profile
            alt.regionTier = other
            guard case let .success(e) = evaluate(packages: packages, profile: alt, pace: pace, day: day) else { continue }
            let delta: Money = gross - e.gross
            if best == nil || Money(abs(delta.value)) > Money(abs(best!.value)) { best = delta }
        }
        return best
    }

    private func nextTier(packages: Int, result: Evaluation, profile: CourierTariffProfile, pace: PaceEstimate,
                          day: String) -> NextTierHint? {
        guard packages > 0 else { return nil }
        let window: Int = Int((Double(packages) * 0.2).rounded(.up))
        guard let progress = result.breakdown.progress.first(where: { $0.unit == .month && $0.target > packages }),
              progress.target - packages <= window,
              case let .success(at) = evaluate(packages: progress.target, profile: profile, pace: pace, day: day) else { return nil }
        return NextTierHint(threshold: progress.target, extraPackages: progress.target - packages,
                            extraNet: at.net - result.net, newRate: progress.reward)
    }
}
```

```swift
import Foundation

/// Bugun karti: aylik plandan gunluk pay ve bugun gerceklesen (secili firmadaki kayitlardan).
struct DailyGoalProgress: Equatable {
    var targetPackages: Int
    var targetHours: Decimal
    var targetNet: Money
    var packages: Int
    var net: Money

    var fraction: Double {
        guard targetPackages > 0 else { return net.value > 0 ? 1 : 0 }
        return min(1, Double(packages) / Double(targetPackages))
    }

    var remainingPackages: Int { max(0, targetPackages - packages) }

    static func today(plan: GoalPlan, entries: [EarningEntry], platform: String, day: String, fuel: FuelCostModel,
                      calendar: Calendar, now: Date) -> DailyGoalProgress {
        let todays: [EarningEntry] = entries.filter { $0.platformCode == platform && calendar.isDate($0.start, inSameDayAs: now) }
        var packages = 0
        var gross: Decimal = 0
        var km: Decimal = 0
        for entry in todays {
            packages += entry.packageCount ?? 0
            gross += Decimal(entry.gross + entry.bonus)
            km += Decimal(entry.distanceKm)
        }
        let net: Money = Money(Money.round(gross)) - fuel.cost(km: km, on: day)
        let days: Int = max(1, plan.workDays)
        let targetNet: Money = Money(Money.round(plan.goal.monthlyTarget().value / Decimal(days)))
        return DailyGoalProgress(targetPackages: plan.packagesPerDay, targetHours: plan.hoursPerDay, targetNet: targetNet,
                                 packages: packages, net: net)
    }
}
```

```swift
import Foundation

/// "Ayni is" girdisi: Firmalar ekraninda her firmanin tarifesiyle hesaplanir.
struct WorkloadSpec: Codable, Equatable {
    var packagesPerDay: Int
    var hoursPerDay: Decimal
    var kmPerPackage: Decimal
    var daysPerMonth: Int

    static func from(pace: PaceEstimate, goal: GoalSpec?) -> WorkloadSpec {
        let days: Int = goal?.monthWorkDays ?? 26
        let hours: Decimal = 8
        let perDay: Int = NSDecimalNumber(decimal: Money.round(pace.packagesPerHour * hours, scale: 0)).intValue
        return WorkloadSpec(packagesPerDay: perDay, hoursPerDay: hours, kmPerPackage: pace.kmPerPackage, daysPerMonth: days)
    }

    var monthlyPackages: Int { packagesPerDay * daysPerMonth }
    var monthlyHours: Decimal { hoursPerDay * Decimal(daysPerMonth) }
    /// Planlayicinin hiz bicimine cevrim.
    var pace: PaceEstimate {
        let rate: Decimal = hoursPerDay > 0 ? Decimal(packagesPerDay) / hoursPerDay : 0
        return PaceEstimate(packagesPerHour: rate, kmPerPackage: kmPerPackage, isAssumed: false, sampleCount: 0)
    }
}

struct FirmComparisonRow: Equatable, Identifiable {
    var platformCode: String
    var isCurrent: Bool
    var gross: Money
    var sideIncome: Money
    var fuel: Money
    var net: Money
    var netPerHour: Money?
    var differenceFromCurrent: Money?
    var versionID: String
    var versionLabel: String
    var effectiveFrom: String
    var assumptions: [String]
    var breakdown: PayBreakdown
    var id: String { platformCode }
}

struct SwitchSuggestion: Equatable {
    var platformCode: String
    var extraNet: Money
    /// Ayni surum ciftinde bir kez bildirmek icin.
    var key: String
}

enum TariffComparison {
    /// Diger firmalar: en dusuk kidem (gecerse bu kademeden baslar); bolge kademesi kendi kademen, o firmada yoksa
    /// sabit ucreti en dusuk kademe. Kendi firman: kendi bilgilerin.
    static func comparisonProfile(for platform: String, own: CourierTariffProfile, catalog: TariffCatalog,
                                  day: String) -> (CourierTariffProfile, [String]) {
        if platform == own.platformCode { return (own, []) }
        var notes: [String] = []
        let versions: [TariffVersion] = catalog.usable(platform: platform, day: day)
        let fixed: [TariffItem] = versions.flatMap { $0.activeItems(on: day) }.filter { $0.type == .fixed }
        var tier: String? = nil
        let tiers: [TariffItem] = fixed.filter { $0.conditions.regionTier != nil }
        if !tiers.isEmpty {
            if let mine = own.regionTier, tiers.contains(where: { $0.conditions.regionTier == mine }) {
                tier = mine
            } else {
                tier = tiers.min { $0.amount < $1.amount }?.conditions.regionTier
            }
        }
        let asksSeniority: Bool = fixed.contains { $0.conditions.minSeniorityYears != nil }
        if asksSeniority { notes.append("yeni başlayan kademesiyle") }
        let vehicle: String? = platform == "getir" ? (own.vehicleType == "car" ? "car" : "motorcycle") : nil
        let profile = CourierTariffProfile(platformCode: platform, regionTier: tier, seniorityYears: 0, vehicleType: vehicle,
                                           flags: own.flags)
        return (profile, notes)
    }

    static func rows(workload: WorkloadSpec, profile own: CourierTariffProfile, catalog: TariffCatalog, fuel: FuelCostModel,
                     day: String) -> [FirmComparisonRow] {
        let planner = GoalPlanner(engine: PayEngine(catalog: catalog), fuel: fuel)
        var rows: [FirmComparisonRow] = []
        for platform in PayPlatform.all where platform.isActive {
            let (profile, notes) = comparisonProfile(for: platform.code, own: own, catalog: catalog, day: day)
            guard case let .success(e) = planner.evaluate(packages: workload.monthlyPackages, profile: profile,
                                                          pace: workload.pace, day: day) else { continue }
            let version: TariffVersion? = catalog.versions.first { $0.id == e.breakdown.tariffVersionID }
            let hours: Decimal = workload.monthlyHours
            rows.append(FirmComparisonRow(
                platformCode: platform.code, isCurrent: platform.code == own.platformCode, gross: e.gross,
                sideIncome: e.sideIncome, fuel: e.fuel, net: e.net,
                netPerHour: hours > 0 ? Money(Money.round(e.net.value / hours)) : nil, differenceFromCurrent: nil,
                versionID: e.breakdown.tariffVersionID, versionLabel: e.breakdown.tariffVersionLabel,
                effectiveFrom: version?.effectiveFrom ?? "", assumptions: notes, breakdown: e.breakdown))
        }
        let current: FirmComparisonRow? = rows.first { $0.isCurrent }
        var others: [FirmComparisonRow] = rows.filter { !$0.isCurrent }.sorted { $1.net < $0.net }
        if let current {
            for index in others.indices { others[index].differenceFromCurrent = others[index].net - current.net }
            return [current] + others
        }
        return others
    }

    /// Baska bir firma kendi firmandan en az `threshold` (yuzde) daha fazla net veriyorsa.
    static func switchSuggestion(rows: [FirmComparisonRow], threshold: Decimal = 5) -> SwitchSuggestion? {
        guard let current = rows.first(where: { $0.isCurrent }), current.net.value > 0 else { return nil }
        let limit: Decimal = current.net.value * (1 + threshold / 100)
        guard let best = rows.filter({ !$0.isCurrent }).max(by: { $0.net < $1.net }), best.net.value >= limit else { return nil }
        return SwitchSuggestion(platformCode: best.platformCode, extraNet: best.net - current.net,
                                key: "\(best.versionID)|\(current.versionID)")
    }
}

/// Son iki yayimlanmis tarife surumu arasindaki degisim ile ayni araliktaki yakit degisimi. Dogrulama durumunu degistirmez.
struct TariffFuelCorrelation: Equatable {
    var platformCode: String
    var fromDay: String
    var toDay: String
    var tariffChange: Decimal
    var fuelChange: Decimal
    var verdict: TariffFuelVerdict

    static let fuelThreshold: Decimal = 3

    /// Karsilastirilan oran: en dusuk esikli paket basi ucret; yoksa en yuksek sabit ucret; yoksa saatlik ucret.
    static func referenceRate(_ version: TariffVersion, day: String) -> Money? {
        let items: [TariffItem] = version.items.filter { !$0.withdrawn }
        let packages: [TariffItem] = items.filter { $0.type == .singlePackage }.sorted { ($0.lowerThreshold ?? 0) < ($1.lowerThreshold ?? 0) }
        if let first = packages.first { return first.amount }
        if let fixed = items.filter({ $0.type == .fixed }).max(by: { $0.amount < $1.amount }) { return fixed.amount }
        return items.first { $0.type == .hourly }?.amount
    }

    static func evaluate(platform: String, catalog: TariffCatalog, fuel: FuelCostModel, today: String) -> TariffFuelCorrelation? {
        guard fuel.hasPrice else { return nil }
        let versions: [TariffVersion] = catalog.versions
            .filter { $0.platformCode == platform && $0.approvalStatus.isUsableForCalculation && $0.effectiveFrom <= today }
            .sorted { ($0.effectiveFrom, $0.versionNumber) < ($1.effectiveFrom, $1.versionNumber) }
        guard versions.count >= 2 else { return nil }
        let old: TariffVersion = versions[versions.count - 2]
        let new: TariffVersion = versions[versions.count - 1]
        guard let a = referenceRate(old, day: old.effectiveFrom), let b = referenceRate(new, day: new.effectiveFrom),
              let tariff = Money.percentChange(from: a, to: b),
              let fuelChange = fuel.change(from: old.effectiveFrom, to: new.effectiveFrom) else { return nil }
        let t: Decimal = Money.round(tariff, scale: 0)
        let f: Decimal = Money.round(fuelChange, scale: 0)
        let verdict: TariffFuelVerdict
        if f >= fuelThreshold {
            verdict = t > 0 ? .consistent : .check
        } else if f <= -fuelThreshold {
            verdict = t > 0 ? .check : .consistent
        } else {
            verdict = .neutral
        }
        return TariffFuelCorrelation(platformCode: platform, fromDay: old.effectiveFrom, toDay: new.effectiveFrom,
                                     tariffChange: t, fuelChange: f, verdict: verdict)
    }
}
```

```swift
struct UnverifiedItem: Hashable, Identifiable {

    var detail: String
    var id: String { detail + "|" + label }

enum UnverifiedItems {
    static func from(_ breakdown: PayBreakdown, platformName: String, includeSide: Bool = true) -> [UnverifiedItem] {
        var lines: [PayLine] = breakdown.lines.filter(\.included)
        if includeSide { lines += breakdown.sideBenefits }
        var result: [UnverifiedItem] = lines.filter { !$0.confidence.isTrusted }.map {
            UnverifiedItem(label: $0.label, detail: "\(platformName) · tarife \(breakdown.tariffVersionLabel)")
        }
        if breakdown.tariffVerification != .verified, result.isEmpty {
            result.append(UnverifiedItem(label: "Tarife sürümü \(breakdown.tariffVersionLabel)",
                                         detail: "\(platformName) · \(breakdown.tariffVerification.displayName)"))
        }
        return merge([result])
    }

    static func from(version: TariffVersion, platformName: String) -> [UnverifiedItem] {
        let items: [UnverifiedItem] = version.items.filter { !$0.withdrawn && !$0.confidence.isTrusted }.map {
            UnverifiedItem(label: $0.label, detail: "\(platformName) · tarife \(version.label)")
        }
        return merge([items])
    }

    /// Sirayi koruyarak tekrarlari atar.
    static func merge(_ lists: [[UnverifiedItem]]) -> [UnverifiedItem] {
        var seen: Set<UnverifiedItem> = []
        var result: [UnverifiedItem] = []
        for item in lists.flatMap({ $0 }) where seen.insert(item).inserted { result.append(item) }
        return result
    }
}
```

```swift
                for _ in 0..<repeats { action() }

        }
    }

    private func stop() {
        timer?.invalidate()
```

```swift
        case .questions: questions.regionTiers.isEmpty || regionTier != nil
        case .fuel: true
        case .goal: true
        }
    }

        Haptics.selection()
        var next = SetupStep(rawValue: step.rawValue + 1) ?? .goal
        if next == .questions, questions.isEmpty { next = .fuel }
        step = next
    }

    private func goBack() {
        var prev = SetupStep(rawValue: step.rawValue - 1) ?? .firm
        if prev == .questions, questions.isEmpty { prev = .firm }
        step = prev

        step = startAt
        let profile = payroll.tariffProfile
        firm = profile?.platformCode
        regionTier = profile?.regionTier
        seniority = profile?.seniorityYears ?? 0
        vehicle = profile?.vehicleType ?? "motorcycle"
        flags = Set(profile?.flags ?? [])
        if let goal = payroll.goal {
            goalAmount = goal.amount.value
            goalPeriod = goal.period
            daysPerWeek = goal.daysPerWeek
            maxHours = NSDecimalNumber(decimal: goal.maxHoursPerDay).intValue
        }
        consumption = payroll.state.fuel.consumptionPer100Km ?? FuelCostModel.defaultConsumption(vehicleType: vehicle)
    }

    private func finish() {
        guard let firm else { return }
        let q = questions
        payroll.chooseFirm(firm, regionTier: q.regionTiers.isEmpty ? nil : regionTier,
                           seniorityYears: q.asksSeniority ? seniority : nil,
                           vehicleType: q.vehicleTypes.isEmpty ? nil : vehicle,
                           flags: Array(flags.intersection(q.flags)))
        if !firmOnly {
            payroll.setConsumption(consumption)
            if let fuelPrice, fuelPrice > 0 { payroll.addFuelPrice(Money(fuelPrice), type: fuelType, on: Date()) }
            if let goalAmount, goalAmount > 0 {
                payroll.setGoal(GoalSpec(amount: Money(goalAmount), period: goalPeriod, daysPerWeek: daysPerWeek,
                                         maxHoursPerDay: Decimal(maxHours)))
            }
        }
        Haptics.success()
        onFinish()
    }
}
```

```swift
                Button("Fiyatı kaydet", action: savePrice)
                    .buttonStyle(PrimaryActionButtonStyle())
                    .disabled((price ?? 0) <= 0)
                    .accessibilityIdentifier("fuelSettings.save")
            } header: { Text("Güncel fiyat") } footer: {
                Text("Fiyat 7 günden eskiyse hatırlatılır. İleride fiyatlar otomatik güncellenecek.")

                    .onChange(of: consumption) { _, v in payroll.setConsumption(v) }
            }
            Section("Fiyat geçmişi") {
                ForEach(payroll.state.fuel.entries.reversed()) { entry in
                    LabeledContent(entry.day, value: "\(PayFormat.tl(entry.pricePerUnit))/\(entry.type.unitName)")
                }
                .onDelete { offsets in
                    let list = Array(payroll.state.fuel.entries.reversed())
                    offsets.map { list[$0].id }.forEach(payroll.deleteFuelPrice)
                }
            }
        }
        .navigationTitle("Araç ve yakıt")
        .saveConfirmation(isPresented: $saved)
        .onAppear {
            consumption = payroll.fuelModel(fallbackCostPerKm: 0).consumption
            type = payroll.state.fuel.entries.last?.type ?? .gasoline
        }
    }

    private func savePrice() {
        guard let price, price > 0 else { return }
        payroll.addFuelPrice(Money(price), type: type, on: date)
        self.price = nil
        withAnimation { saved = true }
    }
```

```swift
import Foundation
import SwiftData

/// Test surumu: secilen senaryonun verisini yazar; olusturulan kimlikleri saklar ki yalnizca onlar silinsin.
@MainActor
struct DemoScenarioLoader {
    let payroll: PayrollModel
    let context: ModelContext
    var defaults: UserDefaults = .standard

    static let backupKey = "demoScenarioBackup"

    var hasRealData: Bool {
        let demo = Set(payroll.state.demoRecordIDs)
        let all = (try? context.fetch(FetchDescriptor<EarningRecord>())) ?? []
        return all.contains { !demo.contains($0.id) }
    }

    func load(_ scenario: DemoScenario, today: Date = .now) {
        clear()
        backupSettings()
        let calendar = Calendar.current
        let todayString = payroll.day(today)
        if let profile = scenario.profile {
            payroll.chooseFirm(profile.platformCode, regionTier: profile.regionTier, seniorityYears: profile.seniorityYears,
                               vehicleType: profile.vehicleType, flags: profile.flags)
        }
        payroll.setGoal(scenario.goal)
        payroll.setConsumption(scenario.consumption)
        for price in scenario.fuelPrices {
            let date = calendar.date(byAdding: .day, value: price.dayOffset, to: today) ?? today
            payroll.addFuelPrice(price.price, type: .gasoline, on: date)
        }
        for version in scenario.userTariffs(today: todayString) {
            _ = payroll.saveUserTariff(version, activate: true)
        }
        var recordIDs: [UUID] = []
        var logIDs: [UUID] = []
        if let profile = scenario.profile {
            for work in scenario.workDays {
                let dayDate = calendar.date(byAdding: .day, value: work.dayOffset, to: calendar.startOfDay(for: today)) ?? today
                let start = calendar.date(byAdding: .hour, value: work.startHour, to: dayDate) ?? dayDate
                let end = start.addingTimeInterval(NSDecimalNumber(decimal: work.hours).doubleValue * 3600)
                var input = PayInput(platformCode: profile.platformCode, startDay: payroll.day(dayDate))
                input.dayPackages = work.packages
                input.totalKm = work.km
                input.hours = work.hours
                input.regionTier = profile.regionTier
                input.seniorityYears = profile.seniorityYears
                input.vehicleType = profile.vehicleType
                input.flags = profile.flags
                if let log = payroll.save(input, method: .quick, start: start, end: end, context: context) {
                    logIDs.append(log.id)
                    if let recordID = log.earningRecordID { recordIDs.append(recordID) }
                }
            }
        }
        payroll.store.setDemoIDs(records: recordIDs, logs: logIDs)
        payroll.reloadState()
        if scenario.resetsOnboarding {
            defaults.set(false, forKey: AppPreferences.hasSeenIntroV17Key)
            defaults.set(false, forKey: AppPreferences.hasCompletedOnboardingKey)
        }
    }

    func clear() {
        let recordIDs = Set(payroll.state.demoRecordIDs)
        let logIDs = Set(payroll.state.demoLogIDs)
        if !recordIDs.isEmpty {
            let records = (try? context.fetch(FetchDescriptor<EarningRecord>())) ?? []
            for record in records where recordIDs.contains(record.id) { context.delete(record) }
            try? context.save()
        }
        payroll.store.removeLogs(ids: logIDs)
        for version in payroll.state.userTariffs where version.id.hasPrefix("demo-") { payroll.deleteUserTariff(version) }
        payroll.store.setDemoIDs(records: [], logs: [])
        restoreSettings()
        payroll.reloadState()
    }

    /// Senaryo oncesi firma/hedef/yakit ayarlarinin yedegi (JSON).
    private func backupSettings() {
        guard defaults.data(forKey: DemoScenarioLoader.backupKey) == nil else { return }
        let s = payroll.state
        let backup = DemoBackup(activePlatform: s.activePlatform, regionTier: s.regionTier, seniorityYears: s.seniorityYears,
                                vehicleType: s.vehicleType, flags: s.flags, goal: s.goal, fuel: s.fuel)
        defaults.set(try? JSONEncoder().encode(backup), forKey: DemoScenarioLoader.backupKey)
    }

    private func restoreSettings() {
        guard let data = defaults.data(forKey: DemoScenarioLoader.backupKey),
              let backup = try? JSONDecoder().decode(DemoBackup.self, from: data) else { return }
        if let platform = backup.activePlatform {
            payroll.chooseFirm(platform, regionTier: backup.regionTier, seniorityYears: backup.seniorityYears,
                               vehicleType: backup.vehicleType, flags: backup.flags)
        }
        payroll.setGoal(backup.goal)
        payroll.store.replaceFuel(backup.fuel)
        defaults.removeObject(forKey: DemoScenarioLoader.backupKey)
    }
}

struct DemoBackup: Codable {
    var activePlatform: String?
    var regionTier: String?
    var seniorityYears: Int?
    var vehicleType: String?
    var flags: [String]
    var goal: GoalSpec?
    var fuel: FuelSettings
}
```

```swift
struct DatabaseInspectorView: View {
    enum Table: String, CaseIterable, Identifiable {
        case earnings = "Kazançlar", expenses = "Giderler", payroll = "Hakediş", fuel = "Yakıt", settings = "Ayarlar", tariff = "Tarife"
        var id: String { rawValue }
    }

    @Environment(PayrollModel.self) private var payroll
    @Query(sort: \EarningRecord.startTime, order: .reverse) private var earnings: [EarningRecord]
    @Query(sort: \ExpenseRecord.date, order: .reverse) private var expenses: [ExpenseRecord]
    @State private var table: Table = .earnings

    var body: some View {
        List {
            Picker("Tablo", selection: $table) {
                ForEach(Table.allCases) { Text($0.rawValue).tag($0) }
            }
            .pickerStyle(.menu)
            Section("\(rows.count) satır") {
                ForEach(Array(rows.enumerated()), id: \.offset) { _, row in
                    NavigationLink {
                        List(row.fields, id: \.0) { field in LabeledContent(field.0, value: field.1) }
                            .navigationTitle(row.title)
                    } label: {
                        VStack(alignment: .leading, spacing: 2) {
                            Text(row.title).font(.subheadline)
                            Text(row.subtitle).font(.caption).foregroundStyle(.secondary)
                        }
                    }
                }
            }
            .accessibilityIdentifier("db.rows")
        }
        .navigationTitle("Veritabanı")
    }

    struct Row {
        var title: String
        var subtitle: String
        var fields: [(String, String)]
    }

    private var rows: [Row] {
        switch table {
        case .earnings:
            return earnings.map { r in
                Row(title: "\(r.platform?.name ?? "—") · \(Formatters.currency(r.grossEarning))",
                    subtitle: r.startTime.formatted(date: .abbreviated, time: .shortened),
                    fields: [("id", r.id.uuidString), ("Başlangıç", r.startTime.formatted()), ("Bitiş", r.endTime.formatted()),

                             ("Paket", r.packageCount.map(String.init) ?? "—"), ("Kaynak", r.sourceRaw),
                             ("Demo", payroll.state.demoRecordIDs.contains(r.id) ? "evet" : "hayır")])

        case .expenses:
            return expenses.map { e in

        case .payroll:
```
